#include "Stack.h"

void stackInit(Stack& s) {
    s.top = NULL;
    s.count = 0;
}

void stackPush(Stack& s, int value) {
    StackNode* node = new StackNode;
    node->data = value;
    node->next = s.top;
    s.top = node;
    s.count++;
}

bool stackPop(Stack& s, int& value) {
    if (s.top == NULL) return false;      // underflow
    StackNode* old = s.top;
    value = old->data;
    s.top = old->next;
    delete old;
    s.count--;
    return true;
}

bool stackPeek(Stack& s, int& value) {
    if (s.top == NULL) return false;
    value = s.top->data;
    return true;
}

bool stackIsEmpty(Stack& s) { return s.top == NULL; }

int stackSize(Stack& s) { return s.count; }

void stackClear(Stack& s) {
    while (s.top != NULL) {
        StackNode* old = s.top;
        s.top = s.top->next;
        delete old;
    }
    s.count = 0;
}

// CS212 Project 94 - Tailoring Order Workshop (simplified course-style version)

#include "Queue.h"

void queueInit(Queue& q) {
    q.front = 0;
    q.count = 0;
    q.capacity = 8;
}

bool queueIsEmpty(Queue& q) { return q.count == 0; }

bool queueIsFull(Queue& q) { return q.count == q.capacity; }

int queueSize(Queue& q) { return q.count; }

bool enqueue(Queue& q, int value) {
    if (queueIsFull(q)) return false;
    int rear = (q.front + q.count) % q.capacity;   // circular: wraps around
    q.items[rear] = value;
    q.count++;
    return true;
}

bool dequeue(Queue& q, int& value) {
    if (queueIsEmpty(q)) return false;
    value = q.items[q.front];
    q.front = (q.front + 1) % q.capacity;
    q.count--;
    return true;
}

bool queueFront(Queue& q, int& value) {
    if (queueIsEmpty(q)) return false;
    value = q.items[q.front];
    return true;
}

int queueAt(Queue& q, int position) {
    return q.items[(q.front + position) % q.capacity];
}

#include <string>
#include "CustomerList.h"

using namespace std;

void listInit(CustomerList& list) {
    list.head = NULL;
    list.tail = NULL;
    list.count = 0;
}

void listClear(CustomerList& list) {
    while (list.head != NULL) {
        CustomerNode* old = list.head;
        list.head = list.head->next;
        delete old;
    }
    list.tail = NULL;
    list.count = 0;
}

Customer* listSearch(CustomerList& list, int id) {
    for (CustomerNode* cur = list.head; cur != NULL; cur = cur->next)
        if (cur->data.id == id) return &cur->data;
    return NULL;
}

bool listInsert(CustomerList& list, Customer& c) {
    if (listSearch(list, c.id) != NULL) return false;   // duplicate ID: reject
    CustomerNode* node = new CustomerNode;
    node->data = c;
    node->next = NULL;
    if (list.head == NULL) {
        list.head = node;
        list.tail = node;
    } else {
        list.tail->next = node;
        list.tail = node;
    }
    list.count++;
    return true;
}

bool listRemove(CustomerList& list, int id) {
    CustomerNode* prev = NULL;
    CustomerNode* cur = list.head;
    while (cur != NULL && cur->data.id != id) {
        prev = cur;
        cur = cur->next;
    }
    if (cur == NULL) return false;            // not found (also covers an empty list)

    if (prev == NULL) list.head = cur->next;  // deleting the first node
    else prev->next = cur->next;

    if (cur == list.tail) list.tail = prev;      // deleting the last node
    delete cur;
    list.count--;
    return true;                                 // only-node case: head = tail = NULL
}

// ----------------------------- merge sort ---------------------------------

// Lower-case copy of a string (so "ahmed" and "Ahmed" sort together).
string lowerName(string s) {
    string r = s;
    for (int i = 0; i < (int)r.size(); i++)
        if (r[i] >= 'A' && r[i] <= 'Z') r[i] = r[i] + ('a' - 'A');
    return r;
}

// true if a must come before b
bool lessThan(Customer& a, Customer& b, int sortKey) {
    if (sortKey == 2 /*by ID*/) return a.id < b.id;
    string na = lowerName(a.name), nb = lowerName(b.name);
    if (na != nb) return na < nb;
    return a.id < b.id;   // tie-break so the order is fully deterministic
}

// Splits the chain in half (walks to the middle); returns the second half.
CustomerNode* splitList(CustomerNode* start) {
    int len = 0;
    for (CustomerNode* p = start; p != NULL; p = p->next) len++;
    CustomerNode* mid = start;
    for (int i = 1; i < len / 2; i++) mid = mid->next;   // last node of the first half
    CustomerNode* second = mid->next;
    mid->next = NULL;
    return second;
}

// Merges two sorted chains with a loop. Takes from 'a' on ties (keeps it stable).
CustomerNode* mergeLists(CustomerNode* a, CustomerNode* b, int sortKey) {
    CustomerNode* result = NULL;   // first node of the merged chain
    CustomerNode* last = NULL;     // last node added so far
    while (a != NULL && b != NULL) {
        CustomerNode* pick;
        if (lessThan(b->data, a->data, sortKey)) {
            pick = b;
            b = b->next;
        } else {
            pick = a;
            a = a->next;
        }
        if (result == NULL) result = pick;
        else last->next = pick;
        last = pick;
    }
    CustomerNode* rest = b;
    if (a != NULL) rest = a;
    if (result == NULL) return rest;
    last->next = rest;
    return result;
}

CustomerNode* mergeSort(CustomerNode* start, int sortKey) {
    if (start == NULL || start->next == NULL) return start;   // 0 or 1 node
    CustomerNode* second = splitList(start);
    start = mergeSort(start, sortKey);
    second = mergeSort(second, sortKey);
    return mergeLists(start, second, sortKey);
}

void listSort(CustomerList& list, int sortKey) {
    list.head = mergeSort(list.head, sortKey);
    list.tail = list.head;                       // re-establish the tail pointer
    while (list.tail != NULL && list.tail->next != NULL) list.tail = list.tail->next;
}

#include "OrderBST.h"

void bstInit(OrderBST& t) {
    t.root = NULL;
    t.count = 0;
}

void destroyNode(BSTNode* n) {
    if (n == NULL) return;
    destroyNode(n->left);       // post-order: children first
    destroyNode(n->right);
    delete n;
}

void bstClear(OrderBST& t) {
    destroyNode(t.root);
    t.root = NULL;
    t.count = 0;
}

// Recursive insert: returns the (possibly new) root of this subtree.
BSTNode* insertNode(BSTNode* root, Order& o, bool& inserted) {
    if (root == NULL) {
        BSTNode* node = new BSTNode;
        node->data = o;
        node->left = NULL;
        node->right = NULL;
        inserted = true;
        return node;
    }
    if (o.id == root->data.id) {          // duplicate ID -> reject
        inserted = false;
        return root;
    }
    if (o.id < root->data.id) root->left = insertNode(root->left, o, inserted);
    else root->right = insertNode(root->right, o, inserted);
    return root;
}

bool bstInsert(OrderBST& t, Order& o) {
    bool inserted = false;
    t.root = insertNode(t.root, o, inserted);
    if (inserted) t.count++;
    return inserted;
}

// Smallest node of a subtree (leftmost). Used as the inorder successor.
BSTNode* getSuccessor(BSTNode* curr) {
    curr = curr->right;
    while (curr != NULL && curr->left != NULL) curr = curr->left;
    return curr;
}

// Recursive delete. Handles: leaf, one child, two children (inorder successor),
// and the root-only tree (returns NULL so the root becomes empty).
BSTNode* deleteNode(BSTNode* root, int id, bool& removed) {
    if (root == NULL) return root;
    if (id < root->data.id) {
        root->left = deleteNode(root->left, id, removed);
    } else if (id > root->data.id) {
        root->right = deleteNode(root->right, id, removed);
    } else {
        removed = true;
        if (root->left == NULL) {          // 0 or 1 child (right)
            BSTNode* temp = root->right;
            delete root;
            return temp;
        }
        if (root->right == NULL) {         // 1 child (left)
            BSTNode* temp = root->left;
            delete root;
            return temp;
        }
        BSTNode* succ = getSuccessor(root);   // 2 children
        root->data = succ->data;
        bool dummy = false;
        root->right = deleteNode(root->right, succ->data.id, dummy);
    }
    return root;
}

bool bstRemove(OrderBST& t, int id) {
    bool removed = false;
    t.root = deleteNode(t.root, id, removed);
    if (removed) t.count--;
    return removed;
}

Order* searchNode(BSTNode* root, int id) {
    if (root == NULL) return NULL;
    if (id == root->data.id) return &root->data;
    if (id > root->data.id) return searchNode(root->right, id);
    return searchNode(root->left, id);
}

Order* bstSearch(OrderBST& t, int id) {
    return searchNode(t.root, id);
}

int heightNode(BSTNode* n) {
    if (n == NULL) return 0;
    int lh = heightNode(n->left), rh = heightNode(n->right);
    if (lh > rh) return 1 + lh;
    return 1 + rh;
}

int bstHeight(OrderBST& t) {
    return heightNode(t.root);
}

// Removes spaces, tabs and line-end characters from both ends of a string.
string trim(string s) {
    int b = 0;
    int e = (int)s.size() - 1;
    while (b <= e && (s[b] == ' ' || s[b] == '\t' || s[b] == '\r' || s[b] == '\n')) b++;
    while (e >= b && (s[e] == ' ' || s[e] == '\t' || s[e] == '\r' || s[e] == '\n')) e--;
    return s.substr(b, e - b + 1);
}

// Shows the prompt and reads one whole line. At end of input the program stops cleanly.
bool ended = false;     // becomes true when the input runs out

bool inputHasEnded() { return ended; }

string readLine(string prompt) {
    if (ended) return "";
    cout << prompt;
    string line;
    if (!getline(cin, line)) {
        cout << "\n(Input ended.)\n";
        ended = true;
        return "";
    }
    // Windows editors/shells may put a UTF-8 byte-order mark at the start of redirected input.
    if (line.size() >= 3 && line.compare(0, 3, "\xEF\xBB\xBF") == 0) line.erase(0, 3);
    return trim(line);
}

string lowerCopy(string s) {
    string r = s;
    for (int i = 0; i < (int)r.size(); i++)
        if (r[i] >= 'A' && r[i] <= 'Z') r[i] = r[i] + ('a' - 'A');
    return r;
}

// Converts text such as "42", "-7" or "+5" to a number. Returns false if the text
// is not a whole number. Very long numbers are clamped (they fail the range check).
bool parseInt(string text, int& value) {
    int i = 0;
    bool negative = false;
    if (i < (int)text.size() && (text[i] == '+' || text[i] == '-')) {
        negative = (text[i] == '-');
        i++;
    }
    if (i >= (int)text.size()) return false;           // no digits
    int v = 0;
    int digits = 0;
    for (; i < (int)text.size(); i++) {
        if (text[i] < '0' || text[i] > '9') return false;
        if (digits < 9) v = v * 10 + (text[i] - '0');
        digits++;
    }
    if (digits > 9) v = 1000000000;           // too big: will fail the range check
    if (negative) v = -v;
    value = v;
    return true;
}

// Converts text such as "42", "42.5" or ".5" to a number. Returns false otherwise.
bool parseDouble(string text, double& value) {
    int i = 0;
    bool negative = false;
    if (i < (int)text.size() && (text[i] == '+' || text[i] == '-')) {
        negative = (text[i] == '-');
        i++;
    }
    double whole = 0.0;
    int digits = 0;
    while (i < (int)text.size() && text[i] >= '0' && text[i] <= '9') {
        if (digits < 15) whole = whole * 10 + (text[i] - '0');
        else whole = 1e15;
        digits++;
        i++;
    }
    double fraction = 0.0;
    double divisor = 1.0;
    int fracDigits = 0;
    if (i < (int)text.size() && text[i] == '.') {
        i++;
        while (i < (int)text.size() && text[i] >= '0' && text[i] <= '9') {
            if (fracDigits < 12) {
                fraction = fraction * 10 + (text[i] - '0');
                divisor = divisor * 10;
            }
            fracDigits++;
            i++;
        }
    }
    if (i != (int)text.size()) return false;           // extra characters
    if (digits + fracDigits == 0) return false;        // no digits at all
    double v = whole + fraction / divisor;
    if (negative) v = -v;
    value = v;
    return true;
}

int readInt(string prompt, int minVal, int maxVal) {
    while (true) {
        string line = readLine(prompt);
        if (ended) return 0;
        int value = 0;
        if (line.empty() || !parseInt(line, value)) {
            cout << "  Error: please enter a whole number.\n";
            continue;
        }
        if (value < minVal || value > maxVal) {
            cout << "  Error: value must be between " << minVal << " and " << maxVal << ".\n";
            continue;
        }
        return value;
    }
}

double readDouble(string prompt, double minVal, double maxVal) {
    while (true) {
        string line = readLine(prompt);
        if (ended) return 0.0;
        double value = 0.0;
        if (line.empty() || !parseDouble(line, value)) {
            cout << "  Error: please enter a number (e.g. 42 or 42.5).\n";
            continue;
        }
        if (value < minVal || value > maxVal) {
            cout << "  Error: value must be between " << minVal << " and " << maxVal << ".\n";
            continue;
        }
        return value;
    }
}

string readNonEmptyLine(string prompt, int maxLen) {
    while (true) {
        string line = readLine(prompt);
        if (ended) return "";
        if (line.empty()) {
            cout << "  Error: this field cannot be empty.\n";
            continue;
        }
        if ((int)line.size() > maxLen) {
            cout << "  Error: maximum length is " << maxLen << " characters.\n";
            continue;
        }
        return line;
    }
}

bool readOptionalLine(string prompt, string& out, int maxLen) {
    while (true) {
        string line = readLine(prompt);
        if (ended) return false;
        if (line.empty()) return false;
        if ((int)line.size() > maxLen) {
            cout << "  Error: maximum length is " << maxLen << " characters.\n";
            continue;
        }
        out = line;
        return true;
    }
}

bool readYesNo(string prompt) {
    while (true) {
        string line = lowerCopy(readLine(prompt));
        if (ended) return false;
        if (line == "y" || line == "yes") return true;
        if (line == "n" || line == "no") return false;
        cout << "  Error: please answer y or n.\n";
    }
}

string intToText(int n) {
    if (n == 0) return "0";
    bool neg = n < 0;
    if (neg) n = -n;
    string digits = "";
    while (n > 0) {
        digits = char('0' + n % 10) + digits;
        n = n / 10;
    }
    if (neg) digits = "-" + digits;
    return digits;
}

// Adds spaces to the right of text until it is 'width' characters wide.
string padRight(string text, int width) {
    string r = text;
    while ((int)r.size() < width) r += " ";
    return r;
}

// Text of a number with a fixed count of decimals, e.g. fixedText(1650, 2) = "1650.00".
// (Only used for values that are never negative.)
string fixedText(double value, int decimals) {
    int scale = 1;
    for (int i = 0; i < decimals; i++) scale = scale * 10;
    int whole = (int)value;
    int part = (int)((value - whole) * scale + 0.5);
    if (part >= scale) {
        whole++;
        part = part - scale;
    }
    string frac = intToText(part);
    while ((int)frac.size() < decimals) frac = "0" + frac;
    return intToText(whole) + "." + frac;
}

         << " chest " << fixedText(c.m.chest, 1) << " | waist " << fixedText(c.m.waist, 1)
         << " | hips " << fixedText(c.m.hips, 1) << " | inseam " << fixedText(c.m.inseam, 1)
         << " | sleeve " << fixedText(c.m.sleeve, 1) << " (cm)\n";
}

// ===================== recursive walks over the order tree =================
// (same pattern as the BST traversal examples: visit left, node, right)

void printInorder(BSTNode* n) {
    if (n == NULL) return;
    printInorder(n->left);
    printOrderRow(n->data);
    printInorder(n->right);
}

void printPreorder(BSTNode* n, int depth) {
    if (n == NULL) return;
    cout << "  " << string(depth * 4, ' ') << "[" << n->data.id << "] "
         << n->data.garment << " (" << statusToString(n->data.status) << ")\n";
    printPreorder(n->left, depth + 1);
    printPreorder(n->right, depth + 1);
}

int countOrdersOf(BSTNode* n, int customerId) {
    if (n == NULL) return 0;
    int here = 0;
    if (n->data.customerId == customerId) here = 1;
    return here + countOrdersOf(n->left, customerId) + countOrdersOf(n->right, customerId);
}

void printOrdersOf(BSTNode* n, int customerId) {
    if (n == NULL) return;
    printOrdersOf(n->left, customerId);
    if (n->data.customerId == customerId) printOrderRow(n->data);
    printOrdersOf(n->right, customerId);
}

struct Totals {
    int waiting;
    int completed;
    int pickedUp;
    double revenue;
    bool customersOk;     // every order's customer exists
};

void collectTotals(BSTNode* n, Totals& t) {
    if (n == NULL) return;
    collectTotals(n->left, t);
    if (n->data.status == 0 /*WAITING*/) t.waiting++;
    else if (n->data.status == 1 /*COMPLETED*/) t.completed++;
    else { t.pickedUp++; t.revenue += n->data.price; }
    if (listSearch(customers, n->data.customerId) == NULL) t.customersOk = false;   // I5
    collectTotals(n->right, t);
}

bool deleteCustomer(int id) {
    if (listSearch(customers, id) == NULL) {
        cout << "  Error: no customer with ID " << id << ".\n";
        return false;
    }
    int n = countOrdersOf(orders.root, id);
    if (n > 0) {
        cout << "  Error: customer #" << id << " still has " << n

        return false;

    listRemove(customers, id);
    cout << "  Customer #" << id << " deleted. Customers on file: " << customers.count << "\n";
    return true;
}

    if (customers.head == NULL) {
        cout << "  No customers to sort.\n";

         << keyName << "):\n";
    for (CustomerNode* cur = customers.head; cur != NULL; cur = cur->next)
        printCustomer(cur->data);
}

// ============== queue / stack helpers (only use the ADT operations) =========

int positionInQueue(int id) {
    for (int i = 0; i < queueSize(sewingQueue); i++)
        if (queueAt(sewingQueue, i) == id) return i + 1;
    return -1;
}

// Removes id from the queue, keeping the order of the others.
bool removeFromQueue(int id) {
    bool found = false;
    int n = queueSize(sewingQueue);
    for (int i = 0; i < n; i++) {
        int x = 0;
        dequeue(sewingQueue, x);
        if (x == id && !found) found = true;      // drop it
        else enqueue(sewingQueue, x);             // put the rest back at the rear
    }
    return found;
}

// Replaces oldId by newId in the queue, in the same position.
bool replaceInQueue(int oldId, int newId) {
    bool found = false;
    int n = queueSize(sewingQueue);
    for (int i = 0; i < n; i++) {
        int x = 0;
        dequeue(sewingQueue, x);
        if (x == oldId) { x = newId; found = true; }
        enqueue(sewingQueue, x);
    }
    return found;
}

// Removes id from the stack: pop onto a temporary stack, then restore.
bool removeFromStack(int id) {
    Stack temp;
    stackInit(temp);
    bool found = false;
    int x = 0;
    while (stackPop(finished, x)) {
        if (x == id) { found = true; break; }
        stackPush(temp, x);
    }
    while (stackPop(temp, x)) stackPush(finished, x);   // restore original order
    return found;
}

// Replaces oldId by newId in the stack, in the same position.
bool replaceInStack(int oldId, int newId) {
    Stack temp;
    stackInit(temp);
    bool found = false;
    int x = 0;
    while (stackPop(finished, x)) {
        if (x == oldId) { stackPush(finished, newId); found = true; break; }
        stackPush(temp, x);
    }
    while (stackPop(temp, x)) stackPush(finished, x);
    return found;
}

bool createOrder(Order& o) {
    if (listSearch(customers, o.customerId) == NULL) {
        cout << "  Error: customer ID " << o.customerId
             << " does not exist. Add the customer first.\n";
        return false;
    }
    if (bstSearch(orders, o.id) != NULL) {
        cout << "  Error: order ID " << o.id << " already exists. Duplicate IDs are rejected.\n";
        return false;
    }
    if (queueIsFull(sewingQueue)) {   // check before touching the BST so nothing is half-done
        cout << "  Error: sewing rack is full (" << sewingQueue.capacity
             << " garments). Sew some garments before accepting new orders.\n";
        return false;
    }
    Order copy = o;
    copy.status = 0 /*WAITING*/;
    bstInsert(orders, copy);
    enqueue(sewingQueue, copy.id);
    cout << "  Order #" << copy.id << " created and placed in the sewing queue (position "
         << queueSize(sewingQueue) << " of " << queueSize(sewingQueue) << ").\n";
    return true;
}

bool showOrder(int id) {
    Order* o = bstSearch(orders, id);
    if (o == NULL) {
        cout << "  Error: no order with ID " << id << ".\n";
        return false;
    }
    printOrderHeader();
    printOrderRow(*o);
    Customer* c = listSearch(customers, o->customerId);
    if (c != NULL) cout << "  Customer: " << c->name << " (" << c->phone << ")\n";
    if (o->status == 0 /*WAITING*/)
        cout << "  Location: sewing queue, position " << positionInQueue(id) << " of "
             << queueSize(sewingQueue) << "\n";
    else if (o->status == 1 /*COMPLETED*/)
        cout << "  Location: finished rack, waiting for pickup\n";
    else
        cout << "  Location: delivered to customer\n";
    return true;
}

bool updateOrder(int id, string* garment, string* alteration,
                 double* price, int* customerId) {
    Order* o = bstSearch(orders, id);
    if (o == NULL) {
        cout << "  Error: no order with ID " << id << ".\n";
        return false;
    }
    if (o->status == 2 /*PICKED_UP*/) {
        cout << "  Error: order #" << id << " was already picked up; archived orders cannot be edited.\n";
        return false;
    }
    if (customerId != NULL && listSearch(customers, *customerId) == NULL) {
        cout << "  Error: customer ID " << *customerId << " does not exist. No changes made.\n";
        return false;
    }
    if (garment != NULL) o->garment = *garment;
    if (alteration != NULL) o->alteration = *alteration;
    if (price != NULL) o->price = *price;
    if (customerId != NULL) o->customerId = *customerId;
    cout << "  Order #" << id << " updated:\n";
    printOrderHeader();
    printOrderRow(*o);
    return true;
}

// Scenario: an order was registered under the wrong ID. The key changes, so the
// BST node must be removed and re-inserted, and the copy of the ID held in the
// queue or stack is rewritten IN PLACE so the order keeps its position.
bool changeOrderId(int oldId, int newId) {
    Order* o = bstSearch(orders, oldId);
    if (o == NULL) {
        cout << "  Error: no order with ID " << oldId << ".\n";
        return false;
    }
    if (oldId == newId) {
        cout << "  Error: the new ID is the same as the old ID. Nothing changed.\n";
        return false;
    }
    if (bstSearch(orders, newId) != NULL) {
        cout << "  Error: order ID " << newId << " is already in use. Duplicate IDs are rejected.\n";
        return false;
    }
    Order copy = *o;
    copy.id = newId;
    bstRemove(orders, oldId);
    bstInsert(orders, copy);

    string where = "BST only (order is picked up)";
    if (copy.status == 0 /*WAITING*/) {
        replaceInQueue(oldId, newId);
        where = "BST + sewing queue (position " + intToText(positionInQueue(newId)) + " kept)";
    } else if (copy.status == 1 /*COMPLETED*/) {
        replaceInStack(oldId, newId);
        where = "BST + finished stack (position kept)";
    }
    cout << "  Order #" << oldId << " is now #" << newId << ". Updated: " << where << ".\n";
    return true;
}

bool deleteOrder(int id) {
    Order* o = bstSearch(orders, id);
    if (o == NULL) {
        cout << "  Error: no order with ID " << id << ".\n";
        return false;
    }
    string where = "BST";
    if (o->status == 0 /*WAITING*/) {
        removeFromQueue(id);
        where += " + sewing queue";
    } else if (o->status == 1 /*COMPLETED*/) {
        removeFromStack(id);
        where += " + finished stack";
    }
    bstRemove(orders, id);
    cout << "  Order #" << id << " deleted from " << where << ". Orders remaining: "
         << orders.count << "\n";
    return true;
}

// =============================== workflow =================================

void showQueue() {
    if (queueIsEmpty(sewingQueue)) {
        cout << "  The sewing queue is empty (0/" << sewingQueue.capacity << ").\n";
        return;
    }
    cout << "  Sewing queue (" << queueSize(sewingQueue) << "/" << sewingQueue.capacity
         << "), front first:\n";
    for (int i = 0; i < queueSize(sewingQueue); i++) {
        int id = queueAt(sewingQueue, i);
        Order* o = bstSearch(orders, id);
        cout << "   " << (i + 1) << ". #" << id;
        if (o != NULL) cout << "  " << o->garment << " - " << o->alteration;
        cout << "\n";
    }
}

bool sewNext() {
    int id = 0;
    if (!dequeue(sewingQueue, id)) {
        cout << "  Nothing to sew: the sewing queue is empty.\n";
        return false;
    }
    Order* o = bstSearch(orders, id);
    if (o == NULL) {   // cannot happen while the invariants hold
        cout << "  Internal error: queued order #" << id << " not found in BST.\n";
        return false;
    }
    o->status = 1 /*COMPLETED*/;
    stackPush(finished, id);
    cout << "  Sewed order #" << id << " (" << o->garment << ": " << o->alteration << ").\n"
         << "  Status -> COMPLETED. Pushed onto finished stack (size " << stackSize(finished)
         << "). Queue now holds " << queueSize(sewingQueue) << ".\n";
    return true;
}

void showFinished() {
    if (stackIsEmpty(finished)) {
        cout << "  No finished alterations are waiting for pickup.\n";
        return;
    }
    cout << "  Recently finished alterations (" << stackSize(finished) << "), most recent first:\n";
    int pos = 1;
    for (StackNode* cur = finished.top; cur != NULL; cur = cur->next) {
        Order* o = bstSearch(orders, cur->data);
        cout << "   " << pos << ". #" << cur->data;
        if (o != NULL) cout << "  " << o->garment << " - " << o->alteration;
        if (pos == 1) cout << "   <- top";
        cout << "\n";
        pos++;
    }
}

// Scenario: quality control rejects the most recently finished garment (LIFO).
bool reworkLatest() {
    int id = 0;
    if (!stackPeek(finished, id)) {
        cout << "  Nothing to rework: the finished stack is empty.\n";
        return false;
    }
    if (queueIsFull(sewingQueue)) {
        cout << "  Error: sewing queue is full; cannot send order #" << id
             << " back for rework right now.\n";
        return false;
    }
    stackPop(finished, id);
    Order* o = bstSearch(orders, id);
    o->status = 0 /*WAITING*/;
    enqueue(sewingQueue, id);
    cout << "  Order #" << id << " failed quality check: popped from finished stack and "
         << "re-queued for sewing (position " << queueSize(sewingQueue) << ").\n";
    return true;
}

bool pickUp(int id) {
    Order* o = bstSearch(orders, id);
    if (o == NULL) {
        cout << "  Error: no order with ID " << id << ".\n";
        return false;
    }
    if (o->status == 0 /*WAITING*/) {
        cout << "  Error: order #" << id << " is not sewn yet (queue position "
             << positionInQueue(id) << ").\n";
        return false;
    }
    if (o->status == 2 /*PICKED_UP*/) {
        cout << "  Error: order #" << id << " was already picked up.\n";
        return false;
    }
    removeFromStack(id);
    o->status = 2 /*PICKED_UP*/;
    cout << "  Order #" << id << " handed to the customer. Amount due: " << fixedText(o->price, 2)
         << " EGP. Status -> PICKED_UP.\n";
    return true;
}

void showSummary() {
    string fullFlag = "";
    if (queueIsFull(sewingQueue)) fullFlag = "  (FULL)";
    Totals t;
    t.waiting = 0;
    t.completed = 0;
    t.pickedUp = 0;
    t.revenue = 0.0;
    t.customersOk = true;
    collectTotals(orders.root, t);
    cout << "  ===== Workshop summary =====\n"
         << "  Customers (linked list) : " << customers.count << "\n"
         << "  Orders (BST)            : " << orders.count << "   tree height " << bstHeight(orders) << "\n"
         << "  Sewing queue            : " << queueSize(sewingQueue) << "/" << sewingQueue.capacity
         << fullFlag << "\n"
         << "  Finished stack          : " << stackSize(finished) << "\n"
         << "  Status counts           : WAITING " << t.waiting << ", COMPLETED " << t.completed
         << ", PICKED_UP " << t.pickedUp << "\n"
         << "  Revenue collected       : " << fixedText(t.revenue, 2) << " EGP\n";
    checkConsistency(true);
}

bool checkConsistency(bool verbose) {
    bool ok = true;
    Totals t;
    t.waiting = 0;
    t.completed = 0;
    t.pickedUp = 0;
    t.revenue = 0.0;
    t.customersOk = true;
    collectTotals(orders.root, t);
    if (!t.customersOk) ok = false;                                                    // I5
    if (t.waiting != queueSize(sewingQueue) || t.completed != stackSize(finished)) ok = false;  // I2, I3
    for (int i = 0; i < queueSize(sewingQueue); i++) {
        Order* o = bstSearch(orders, queueAt(sewingQueue, i));
        if (o == NULL || o->status != 0 /*WAITING*/) ok = false;
    }
    for (StackNode* cur = finished.top; cur != NULL; cur = cur->next) {
        Order* o = bstSearch(orders, cur->data);
        if (o == NULL || o->status != 1 /*COMPLETED*/) ok = false;
    }
    if (verbose) {
        string verdict = "FAILED";
        if (ok) verdict = "OK";
        cout << "  Consistency check       : " << verdict << "\n";
    }
    return ok;
}

Measurements readMeasurements() {
    Measurements m;
    m.chest  = readDouble("  Chest (cm, 30-200): ", 30, 200);
    m.waist  = readDouble("  Waist (cm, 30-200): ", 30, 200);
    m.hips   = readDouble("  Hips (cm, 30-200): ", 30, 200);
    m.inseam = readDouble("  Inseam (cm, 30-130): ", 30, 130);
    m.sleeve = readDouble("  Sleeve (cm, 30-100): ", 30, 100);   // if input ended, the caller checks inputHasEnded()
    return m;
}

int readId(string what) {
    return readInt("  " + what + " (1-99999): ",
                   1, 99999);
}

void menuAddCustomer() {
    Customer c;
    c.id = readId("Customer ID");
    if (inputHasEnded()) return;
    if (findCustomer(c.id) != NULL) {   // fail fast before asking for everything
        cout << "  Error: customer ID " << c.id << " already exists. Duplicate IDs are not allowed.\n";
        return;
    }
    c.name = readNonEmptyLine("  Name: ", 40);
    if (inputHasEnded()) return;
    c.phone = readNonEmptyLine("  Phone: ", 20);
    if (inputHasEnded()) return;
    c.m = readMeasurements();
    if (inputHasEnded()) return;
    addCustomer(c);
}

void menuUpdateCustomer() {
    int id = readId("Customer ID to update");
    if (inputHasEnded()) return;
    Customer* c = findCustomer(id);
    if (c == NULL) {
        cout << "  Error: no customer with ID " << id << ".\n";
        return;
    }
    cout << "  Press Enter to keep the current value.\n";
    string s;
    if (readOptionalLine("  Name [" + c->name + "]: ", s, 40)) c->name = s;
    if (inputHasEnded()) return;
    if (readOptionalLine("  Phone [" + c->phone + "]: ", s, 20)) c->phone = s;
    if (inputHasEnded()) return;
    bool retake = readYesNo("  Re-take measurements? (y/n): ");
    if (inputHasEnded()) return;
    if (retake) {
        Measurements m = readMeasurements();
        if (inputHasEnded()) return;
        c->m = m;
    }
    cout << "  Customer #" << id << " updated.\n";
    showCustomer(id);
}

void menuCreateOrder() {
    Order o;
    o.id = readId("Order ID");
    if (inputHasEnded()) return;
    if (findOrder(o.id) != NULL) {
        cout << "  Error: order ID " << o.id << " already exists. Duplicate IDs are rejected.\n";
        return;
    }
    o.customerId = readId("Customer ID");
    if (inputHasEnded()) return;
    if (findCustomer(o.customerId) == NULL) {
        cout << "  Error: customer ID " << o.customerId << " does not exist. Add the customer first.\n";
        return;
    }
    o.garment = readNonEmptyLine("  Garment: ", 40);
    if (inputHasEnded()) return;
    o.alteration = readNonEmptyLine("  Alteration needed: ", 60);
    if (inputHasEnded()) return;
    o.price = readDouble("  Price (EGP, 0-100000): ", 0, 100000);
    if (inputHasEnded()) return;

}

void menuUpdateOrder() {
    int id = readId("Order ID to update");
    if (inputHasEnded()) return;
    Order* o = findOrder(id);
    if (o == NULL) {
        cout << "  Error: no order with ID " << id << ".\n";
        return;
    }
    cout << "  Press Enter to keep the current value.\n";
    string garment, alteration, priceText, custText;
    bool g = readOptionalLine("  Garment [" + o->garment + "]: ", garment, 40);
    if (inputHasEnded()) return;
    bool a = readOptionalLine("  Alteration [" + o->alteration + "]: ", alteration, 60);
    if (inputHasEnded()) return;
    double price = o->price;
    bool p = readYesNo("  Change price? (y/n): ");
    if (inputHasEnded()) return;
    if (p) price = readDouble("  New price (EGP, 0-100000): ", 0, 100000);
    if (inputHasEnded()) return;
    int cust = o->customerId;
    bool c = readYesNo("  Re-assign to another customer? (y/n): ");
    if (inputHasEnded()) return;
    if (c) cust = readId("New customer ID");
    if (inputHasEnded()) return;
    string* gp = NULL;
    string* ap = NULL;
    double* pp = NULL;
    int* cp = NULL;
    if (g) gp = &garment;
    if (a) ap = &alteration;
    if (p) pp = &price;
    if (c) cp = &cust;
    updateOrder(id, gp, ap, pp, cp);
}

void handleChoice(int choice) {
    switch (choice) {
        case 1: menuAddCustomer(); break;
        case 2: listCustomers(); break;
        case 3: {
            int id = readId("Customer ID");
            if (inputHasEnded()) break;
            showCustomer(id);
            break;
        }
        case 4: menuUpdateCustomer(); break;
        case 5: {
            int id = readId("Customer ID to delete");
            if (inputHasEnded()) break;
            deleteCustomer(id);
            break;
        }
        case 6: {
            int k = readInt("  Sort by 1) name  2) ID: ", 1, 2);
            if (inputHasEnded()) break;
            if (k == 1) sortCustomers(1 /*by name*/);
            else sortCustomers(2 /*by ID*/);
            break;
        }
        case 7: menuCreateOrder(); break;
        case 8: {
            int id = readId("Order ID");
            if (inputHasEnded()) break;
            showOrder(id);
            break;
        }
        case 9: listOrdersInorder(); break;
        case 10: showTreePreorder(); break;
        case 11: menuUpdateOrder(); break;
        case 12: {
            int oldId = readId("Current (wrong) order ID");
            if (inputHasEnded()) break;
            if (findOrder(oldId) == NULL) {
                cout << "  Error: no order with ID " << oldId << ".\n";
                break;
            }
            int newId = readId("Correct order ID");
            if (inputHasEnded()) break;
            changeOrderId(oldId, newId);
            break;
        }
        case 13: {
            int id = readId("Order ID to delete");
            if (inputHasEnded()) break;
            if (findOrder(id) == NULL) {
                cout << "  Error: no order with ID " << id << ".\n";
                break;
            }
            bool sure = readYesNo("  Really delete order #" + intToText(id) + "? (y/n): ");
            if (inputHasEnded()) break;
            if (sure)
                deleteOrder(id);
            else
                cout << "  Delete cancelled.\n";
            break;
        }
        case 14: showQueue(); break;
        case 15: sewNext(); break;
        case 16: showFinished(); break;
        case 17: reworkLatest(); break;
        case 18: {
            int id = readId("Order ID being picked up");
            if (inputHasEnded()) break;
            pickUp(id);
            break;
        }
        case 19: showSummary(); break;
        case 20: loadSampleData(); break;
        default: break;
    }
}

int main() {
    initWorkshop();
    cout << "Welcome to the Tailoring Order Workshop (CS212 Project 94).\n"
         << "Tip: choose 20 to load sample data.\n";
    while (true) {
        printMenu();
        int choice = readInt("Choose an option (0-20): ", 0, 20);
        if (inputHasEnded() || choice == 0) break;
        handleChoice(choice);
        if (inputHasEnded()) break;
    }
    cout << "Goodbye.\n";
    closeWorkshop();
    return 0;
}
